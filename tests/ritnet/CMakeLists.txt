cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(illixr_ritnet_standalone)
option(ILLIXR_BATCH_TRACE "Export deferred diagnostic records through batched HTIF writes" ON)
target_compile_definitions(app PRIVATE ILLIXR_BATCH_TRACE=$<BOOL:${ILLIXR_BATCH_TRACE}>)
get_filename_component(ILLIXR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ILLIXR_ROOT}/cmake/illixr_openblas.cmake")
include("${ILLIXR_ROOT}/cmake/illixr_ritnet.cmake")
set(ILLIXR_RITNET_REFERENCE "" CACHE PATH "Generated exact reference output header directory")
if(NOT EXISTS "${ILLIXR_RITNET_REFERENCE}/ritnet_reference.h")
 message(FATAL_ERROR "Standalone RITNet requires a preserved host-reference tensor")
endif()
set(ILLIXR_CORE_HZ 1000000000 CACHE STRING "Modeled CPU frequency")
target_compile_definitions(app PRIVATE ILLIXR_GPU_PIPELINE=0 ILLIXR_CORE_HZ=${ILLIXR_CORE_HZ} EIGEN_DONT_VECTORIZE EIGEN_DONT_PARALLELIZE)
target_include_directories(app PRIVATE "${ILLIXR_ROOT}/src" "${ZEPHYR_BASE}/../modules/lib/eigen" "${ILLIXR_RITNET_REFERENCE}")
target_compile_options(app PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:SHELL:-include ${ILLIXR_ROOT}/src/helper/eigen_lib_fix.hpp>")
target_sources(app PRIVATE main.cpp "${ILLIXR_ROOT}/src/eye_tracking.cpp" "${ILLIXR_ROOT}/src/relative_clock.cpp" "${ILLIXR_ROOT}/src/trace_transport.cpp" "${ILLIXR_ROOT}/plugins/opencv/src/atomic_shim.c")

option(ILLIXR_RITNET_VECTOR_PREFLIGHT "Run vector platform checks for FPGA preflight" OFF)
if(ILLIXR_RITNET_VECTOR_PREFLIGHT)
 target_compile_definitions(app PRIVATE ILLIXR_RITNET_VECTOR_PREFLIGHT=1)
endif()
