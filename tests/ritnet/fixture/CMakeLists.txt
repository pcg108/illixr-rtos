cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(ritnet_isolated_operation)
set(RITNET_FIXTURE_DIR "" CACHE PATH "Output of extract_ritnet_fixture.py")
if(NOT EXISTS "${RITNET_FIXTURE_DIR}/fixture.c")
  message(FATAL_ERROR "An extracted, verified RITNet fixture is required")
endif()
target_sources(app PRIVATE main.c "${RITNET_FIXTURE_DIR}/fixture.c")
target_include_directories(app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../third_party/ritnet/port")
target_compile_definitions(app PRIVATE BAREMETAL=1)
target_compile_options(app PRIVATE -fno-fast-math -ffp-contract=off)
