cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(illixr_gemmini_standalone)
option(ILLIXR_GEMMINI_EDGE_CASES_ONLY "Supplemental BLAS boundary fixtures; not the full platform gate" OFF)
get_filename_component(ILLIXR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ILLIXR_ROOT}/cmake/illixr_openblas.cmake")
set(ILLIXR_GEMMINI_ORACLE_MODE "runtime" CACHE STRING "Standalone oracle: runtime, capture, verify or table")
set_property(CACHE ILLIXR_GEMMINI_ORACLE_MODE PROPERTY STRINGS runtime capture verify table)
set(ILLIXR_GEMMINI_ORACLE_CAPTURE "" CACHE FILEPATH "Successful standalone Spike reference capture")
set(ILLIXR_GEMMINI_MAIN "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
if(NOT ILLIXR_GEMMINI_ORACLE_MODE STREQUAL "runtime")
  if(NOT ILLIXR_GEMMINI_ORACLE_MODE MATCHES "^(capture|verify|table)$" OR ILLIXR_GEMMINI_EDGE_CASES_ONLY)
    message(FATAL_ERROR "Reference tables require a complete standalone fixture and a valid mode")
  endif()
  set(oracle_dir "${CMAKE_CURRENT_BINARY_DIR}/fixture_oracle")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${ILLIXR_ROOT}/scripts/prepare_gemmini_fixture_oracle.py"
    "${ILLIXR_ROOT}/src/gemmini_selftest.cpp"
    "${ILLIXR_ROOT}/src/blas_reference.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/fixture_oracle.hpp")
  set(oracle_args --root "${ILLIXR_ROOT}" --output "${oracle_dir}" --mode "${ILLIXR_GEMMINI_ORACLE_MODE}")
  if(NOT ILLIXR_GEMMINI_ORACLE_MODE STREQUAL "capture")
    list(APPEND oracle_args --capture "${ILLIXR_GEMMINI_ORACLE_CAPTURE}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ILLIXR_GEMMINI_ORACLE_CAPTURE}")
  endif()
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${ILLIXR_ROOT}/scripts/prepare_gemmini_fixture_oracle.py" ${oracle_args}
                  RESULT_VARIABLE oracle_result)
  if(NOT oracle_result EQUAL 0)
    message(FATAL_ERROR "Failed to prepare standalone oracle sources")
  endif()
  get_target_property(original_sources app SOURCES)
  foreach(source IN LISTS original_sources)
    if(source MATCHES "/(gemmini_selftest|blas_reference)\\.cpp$")
      list(REMOVE_ITEM original_sources "${source}")
    endif()
  endforeach()
  set_property(TARGET app PROPERTY SOURCES "${original_sources}")
  target_sources(app PRIVATE "${oracle_dir}/gemmini_selftest.cpp" "${oracle_dir}/blas_reference.cpp")
  if(NOT ILLIXR_GEMMINI_ORACLE_MODE STREQUAL "capture")
    target_sources(app PRIVATE "${oracle_dir}/fixture_oracle_data.cpp")
  endif()
  string(TOUPPER "${ILLIXR_GEMMINI_ORACLE_MODE}" oracle_mode_upper)
  target_compile_definitions(app PRIVATE "ILLIXR_ORACLE_${oracle_mode_upper}=1")
  target_include_directories(app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  set(ILLIXR_GEMMINI_MAIN "${oracle_dir}/main.cpp")
endif()
if(NOT ILLIXR_LINALG_BACKEND STREQUAL "openblas_gemmini_fp32")
  message(FATAL_ERROR "This standalone accelerator fixture requires Gemmini")
endif()
target_include_directories(app PRIVATE "${ILLIXR_ROOT}/src" "${ZEPHYR_BASE}/../modules/lib/eigen")
target_compile_options(app PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:SHELL:-include ${ILLIXR_ROOT}/src/helper/eigen_lib_fix.hpp>")
target_compile_definitions(app PRIVATE EIGEN_DONT_PARALLELIZE EIGEN_DONT_VECTORIZE)
set(ILLIXR_CORE_HZ "0" CACHE STRING "Modeled core frequency; zero for functional Spike")
target_compile_definitions(app PRIVATE ILLIXR_CORE_HZ=${ILLIXR_CORE_HZ})
target_sources(app PRIVATE "${ILLIXR_GEMMINI_MAIN}" "${ILLIXR_ROOT}/src/trace_transport.cpp"
                           "${ILLIXR_ROOT}/plugins/opencv/src/atomic_shim.c")
if(ILLIXR_GEMMINI_EDGE_CASES_ONLY)
  target_compile_definitions(app PRIVATE ILLIXR_GEMMINI_EDGE_CASES_ONLY=1)
endif()

option(ILLIXR_PACKING_TESTS "Compare scalar and RVV packing bit for bit" OFF)
option(ILLIXR_PACKING_TESTS_ONLY "Run only vector preflight and packing equivalence" OFF)
if(ILLIXR_PACKING_TESTS)
  if(NOT ILLIXR_GEMMINI_PACKING STREQUAL "rvv")
    include("${ILLIXR_ROOT}/cmake/illixr_rvv_packing.cmake")
    illixr_add_rvv_packing(app)
  endif()
  target_sources(app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/packing_cases.cpp")
  target_compile_definitions(app PRIVATE ILLIXR_PACKING_TESTS=1)
  if(ILLIXR_PACKING_TESTS_ONLY)
    target_compile_definitions(app PRIVATE ILLIXR_PACKING_TESTS_ONLY=1)
  endif()
elseif(ILLIXR_PACKING_TESTS_ONLY)
  message(FATAL_ERROR "Packing-only requires packing tests")
endif()

option(ILLIXR_PACKING_BENCHMARK "Paired scalar/RVV packing timings" OFF)
if(ILLIXR_PACKING_BENCHMARK)
  if(NOT ILLIXR_PACKING_TESTS)
    message(FATAL_ERROR "Packing benchmark requires equivalence tests")
  endif()
  target_compile_definitions(app PRIVATE ILLIXR_PACKING_BENCHMARK=1)
endif()

option(ILLIXR_PACKING_DIAGNOSTICS "Record conversion flag timing without relaxing checks" OFF)
if(ILLIXR_PACKING_DIAGNOSTICS)
  if(NOT ILLIXR_PACKING_TESTS)
    message(FATAL_ERROR "Packing diagnostics require packing tests")
  endif()
  target_compile_definitions(app PRIVATE ILLIXR_PACKING_DIAGNOSTICS=1)
endif()
